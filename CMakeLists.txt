cmake_minimum_required(VERSION 3.16)
project(openvq VERSION 0.1.0 LANGUAGES CXX)
option(OPENVQ_BUILD_TESTS "Build OpenVQ tests" ON)
option(OPENVQ_BUILD_CLI "Build OpenVQ CLI" ON)

add_library(openvq STATIC src/openvq.cpp src/preprocessing.cpp src/calibration.cpp src/advanced.cpp src/hybrid.cpp src/phase4.cpp src/trace.cpp)
target_include_directories(openvq PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(openvq PUBLIC cxx_std_17)
if(MSVC)
  target_compile_options(openvq PRIVATE /W4)
else()
  target_compile_options(openvq PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(OPENVQ_BUILD_CLI)
  add_executable(openvq_cli src/cli.cpp)
  target_link_libraries(openvq_cli PRIVATE openvq)
  add_executable(openvq_trace_cli src/trace_cli.cpp)
  target_link_libraries(openvq_trace_cli PRIVATE openvq)
endif()

if(OPENVQ_BUILD_TESTS)
  enable_testing()
  add_executable(openvq_tests tests/test_openvq.cpp)
  target_link_libraries(openvq_tests PRIVATE openvq)
  add_test(NAME openvq_tests COMMAND openvq_tests)
  add_executable(openvq_advanced_tests tests/test_advanced.cpp)
  target_link_libraries(openvq_advanced_tests PRIVATE openvq)
  add_test(NAME openvq_advanced_tests COMMAND openvq_advanced_tests)
  add_executable(openvq_hybrid_tests tests/test_hybrid.cpp)
  target_link_libraries(openvq_hybrid_tests PRIVATE openvq)
  add_test(NAME openvq_hybrid_tests COMMAND openvq_hybrid_tests)
  add_executable(openvq_preprocessing_tests tests/test_preprocessing.cpp)
  target_link_libraries(openvq_preprocessing_tests PRIVATE openvq)
  add_test(NAME openvq_preprocessing_tests COMMAND openvq_preprocessing_tests)
  add_executable(openvq_phase4_tests tests/test_phase4.cpp)
  target_link_libraries(openvq_phase4_tests PRIVATE openvq)
  add_test(NAME openvq_phase4_tests COMMAND openvq_phase4_tests)
  add_executable(openvq_require_failure tests/test_require_failure.cpp)
  add_test(NAME openvq_require_failure COMMAND openvq_require_failure)
  set_tests_properties(openvq_require_failure PROPERTIES WILL_FAIL TRUE)
endif()
